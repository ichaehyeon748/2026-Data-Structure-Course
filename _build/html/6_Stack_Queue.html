from DataStructureCode.SimpleLinkedList import *

class LinkedStack:
	def __init__(self):
		self.stack = SimpleLinkedList()

	def push(self, x):
		self.stack.insert(0, x)

	def pop(self):
		return self.stack.pop(0)

	def top(self):
		if self.isEmpty():
			return None
		else:
			return self.stack.get(0)

	def isEmpty(self) -> bool:
		return self.stack.isEmpty()

	def size(self):
		return self.stack.size()

	def printStack(self):
		print("Elements from top to bottom: ", end=" ")
		for i in range(self.stack.size()):
			print(self.stack.get(i), end=" ")

	def reverse(self):
		temp_stack = []
		while not self.isEmpty():
			temp_stack.append(self.pop())
	  for i in range(len(temp_stack)):
			self.push(temp_stack[i])

Reverse: Queue#

  1. Create a function that reverses individual characters in a queue using recursive approach. Note that the original queue should be reversed.

  2. Try to add reverse() method to the ListQueue class.

from DataStructureCode.LinkedQueue import *
queue = ListQueue()
queue.enqueue("Mon"); queue.enqueue("Tue"); queue.enqueue("Wed"); queue.enqueue("Thurs"); queue.enqueue("Fri")
queue.printQueue()

# A function that reverses individual characters in a queue using recursive approach. Note that the original queue should be reversed.
def reverseQueue(queue):
    if queue.isEmpty():
        return
    # take out (dequeue) from front and put it back (enqueue) to rear
    item = queue.dequeue()
    # using recursion algorithm
    reverseQueue(queue)
    queue.enqueue(item)

reverseQueue(queue)
print("After Reversing the queue:")
queue.printQueue()

# add the below method to the ListQueue class.
def reverse(self):
		if self.isEmpty():
			return
		item = self.dequeue()
		reverse(self)
		self.enqueue(item)
