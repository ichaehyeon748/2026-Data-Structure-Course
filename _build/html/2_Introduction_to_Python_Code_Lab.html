class Attendance:
  def __init__(self, id, name, department,attendance):
    self.id = id
    self.name = name
    self.department = department
    self.attendance = attendance

  def mark_attendance(self, date, status):
    self.attendance.append({"date": date, "status": status})

  def view_attendance(self):
    for entry in self.attendance:
      if entry["status"] == 1:
        print(f"date: {entry["date"]} | status: present")
      else:
        print(f"date: {entry["date"]} | status: absent")

  def update_attendance(self, date, status):
    for entry in self.attendance:
      if entry["date"] == date:
        entry["status"] = status
        return
      else:
        print("Date not found in attendance records.")
